#!/bin/sh
set -eu
D=$(CDPATH= cd -- "$(dirname -- "$0")" && pwd)
. "$D/env.sh"
# Compile both binaries before running this script. Run only after all task benchmarks/tests pause.
# Every sample is a new process. Streamed SHA256 warms the file cache before measurement.
# No Log/Close/digest operation is included in elapsed time or allocation deltas.
set -C
sha256sum "$D/growing1000.jsonl" > "$D/fixture-before.sha256"
for variant in baseline current; do
  for sample in 1 2 3; do
    "$D/open-$variant" "$D/growing1000.jsonl" > "$D/$variant-$sample.json"
  done
done
sha256sum "$D/growing1000.jsonl" > "$D/fixture-after.sha256"
cmp "$D/fixture-before.sha256" "$D/fixture-after.sha256"
